\section{Dataset \& ML}
\begin{frame}{2. Dataset \& Feature Engineering}
    \framesubtitle{Rubric: Methodology -- Datasets \& Preprocessing}
    \begin{itemize}
        \item \textbf{Unified Research Dataset:} \textbf{23{,}970 records} across 26 attributes, combining:
        \begin{itemize}
            \item \textbf{VeReMi Reference (Primary, 72.9\%):} 17{,}470 records from authentic 802.11p BSM simulations in Luxembourg LuST urban scenario (Types 1, 2, 4, 8, 16 across low/med/high densities).
            \item \textbf{EdgeTrust-VANET (Secondary, 20.9\%):} 5{,}000 records (DoS, Sybil, Blackhole, Packet Injection).
            \item \textbf{V-RADD Augmentation (6.3\%):} 1{,}500 records of Grayhole (selective forwarding) and Replay.
        \end{itemize}
        \item \textbf{Class Distribution:} 70.6\% Benign (16{,}917) vs. 29.4\% Malicious (7{,}053) -- realistic attack density.
        \item \textbf{14 Real-Time, Leakage-Free Features (RSU-Observable):}
        \begin{itemize}
            \item \textbf{Mobility (5):} \texttt{position\_x}, \texttt{position\_y}, \texttt{speed}, \texttt{direction}, \texttt{acceleration}
            \item \textbf{Network (6):} \texttt{packet\_sent}, \texttt{packet\_received}, \texttt{packet\_drop\_ratio}, \texttt{latency}, \texttt{retransmission\_count}, \texttt{signal\_strength} (dBm)
            \item \textbf{Trust (3):} \texttt{trust\_score}, \texttt{neighbor\_trust\_score\_avg}, \texttt{historical\_trust\_score}
        \end{itemize}
        \item \textbf{Leakage Prevention \& Group Splitting:}
        \begin{itemize}
            \item \textbf{Removed} 4 retrospective counters (\texttt{false\_packet\_injection}, \texttt{blackhole\_attack\_attempts}, etc.) -- audit proves they cause real-time model collapse (100\% $\rightarrow$ 86.05\% F1).
            \item Partitioned via \textbf{Group-based vehicle splitting} (\texttt{node\_id}) with \textbf{zero vehicle/trajectory overlap} across Train (70\%), Validation (15\%), and Holdout Test (15\%).
        \end{itemize}
        \item \textbf{Result:} Genuine, deployable \textbf{97.74\% Weighted F1} on unseen trajectories, and \textbf{90.99\% Live Accuracy} in OMNeT++/Veins RSU deployment.
    \end{itemize}
\end{frame}

        \vspace{0.2cm}
        \textbf{Heuristic Trust Update (EMA):}
        \[
        T_t = \alpha \cdot T_{t-1} + (1-\alpha)\cdot E_t
        \]
        {\scriptsize $T_t$: current trust, $T_{t-1}$: historical trust, \\ $\alpha = 0.70$ (memory retention factor), \\ $E_t \in [0, 1]$: observable real-time evidence:
        \[
        E_t = 0.35 E_{\text{mobility}} + 0.30 E_{\text{delivery}} + 0.20 E_{\text{latency}} + 0.15 E_{\text{channel}}
        \]
        Spatial neighbor trust computed via 300m KD-Tree.}
    \end{columns}
\end{frame}

\section{Technical Knowledge}
\begin{frame}{10. Technical Knowledge \& Design Decisions}
    \framesubtitle{Rubric: Understanding of Methodology, Implementation \& Design Choices}
    \begin{itemize}
        \item \textbf{Why Tree Ensembles Win in Live VANETs:} Physical vehicular telemetry exhibits hard non-linear boundaries (e.g. packet drop ratio $> 0.40$, trust $< 0.40$). Bagging in Random Forest aggregates 150 orthogonal trees, effectively filtering non-Gaussian radio noise and achieving the lowest missed attack rate (\textbf{6.89\%}).
        \item \textbf{Eliminating Trajectory Leakage:} Random row splitting inflates scores (99.29\% F1) due to temporal autocorrelation. Enforcing \textbf{Group-based vehicle splitting (\texttt{node\_id})} guarantees zero vehicle overlap and honest generalization.
        \item \textbf{Why $\alpha = 0.70$ in EMA Trust Engine:} Balances historical vehicle stability with rapid anomaly reaction; persistently penalizes attackers within 4 packets without overreacting to single-packet sensor jitter.
        \item \textbf{Two-Tier RSU Hybrid Architecture:}
        \begin{itemize}
            \item Tier 1: Real-time heuristic trust engine executes in $O(1)$ time on every received 802.11p beacon.
            \item Tier 2: 14-feature ML model predicts transmission integrity.
            \item \textbf{Safety Policy:} Requires \textbf{both} low trust ($< 0.40$) \textbf{and} ML malicious prediction to \textbf{BLOCK}, ensuring legitimate traffic is never falsely disconnected.
        \end{itemize}
        \item \textbf{Class Imbalance \& Validation Protocol:} Handled 70.6\% normal vs 29.4\% malicious distribution via Weighted F1 optimization and 5-Fold Stratified Group Cross-Validation (\textbf{96.95\% $\pm$ 1.12\%}).
    \end{itemize}
\end{frame}
